// 工程说明：本段集中处理状态、边界或诊断，避免跨周期残留。
// 边界说明：执行前后保持输出和错误码可被在线诊断追踪。
// 约束：HTTP/1.1 请求必须校验 Host，缺失或重复 Host 直接按协议错误处理，避免虚拟主机场景路由歧义。
// 约束：Transfer-Encoding 与 Content-Length 同时存在必须拒绝，避免请求走私类歧义进入 PLC Server。
// 风险：Content-Length、chunked 和 close 语义互斥处理，任何长度错误都必须停在协议层。
// 诊断：解析失败统一输出 eError 和 sDiagMsg，外部 Server 探针可把 4xx 响应映射到具体协议原因。

M_ParseRequest := FALSE;
M_ResetRequest(
    stRequest := stRequest
    );

iFirstLineEnd := FIND(sMessage, '$R$N');
iHeaderEnd := FIND(sMessage, '$R$N$R$N');
IF (iFirstLineEnd <= 1) OR (iHeaderEnd <= iFirstLineEnd) THEN
    M_SetError(
        eNewError := E_HttpError.iNeedMoreData,
        sMessage  := 'request header is incomplete'
        );
    RETURN;
END_IF

sStartLine := LEFT(sMessage, iFirstLineEnd - 1);
iSpace1 := FIND(sStartLine, ' ');
IF iSpace1 <= 1 THEN
    M_SetError(
        eNewError := E_HttpError.iInvalidStartLine,
        sMessage  := 'request start-line misses method'
        );
    RETURN;
END_IF

sAfterMethod := MID(sStartLine, LEN(sStartLine) - iSpace1, iSpace1 + 1);
iSpace2Relative := FIND(sAfterMethod, ' ');
IF iSpace2Relative <= 1 THEN
    M_SetError(
        eNewError := E_HttpError.iInvalidStartLine,
        sMessage  := 'request start-line misses version'
        );
    RETURN;
END_IF

stRequest.sMethod := LEFT(sStartLine, iSpace1 - 1);
stRequest.sTarget := LEFT(sAfterMethod, iSpace2Relative - 1);
stRequest.sVersion := MID(sAfterMethod, LEN(sAfterMethod) - iSpace2Relative, iSpace2Relative + 1);
stRequest.eMethod := M_ParseMethod(
    sMethod := stRequest.sMethod
    );

IF (LEN(stRequest.sTarget) = 0) OR (FIND(stRequest.sVersion, 'HTTP/') <> 1) THEN
    M_SetError(
        eNewError := E_HttpError.iInvalidStartLine,
        sMessage  := 'request target or version is invalid'
        );
    RETURN;
END_IF

iHeaderStart := iFirstLineEnd + 2;
iHeaderLen := iHeaderEnd - iHeaderStart;
IF iHeaderLen > GVL_Http.cnMaxHeaderSize THEN
    M_SetError(
        eNewError := E_HttpError.iBufferTooSmall,
        sMessage  := 'request header exceeds limit'
        );
    RETURN;
END_IF
IF iHeaderLen > 0 THEN
    sHeaderText := MID(sMessage, iHeaderLen, iHeaderStart);
ELSE
    sHeaderText := '';
END_IF
stRequest.sRawHeaders := sHeaderText;

bHeaderFound := F_HttpFindHeader(
    sHeaders        := sHeaderText,
    sHeaderName     := 'Host',
    sValue          => sHeaderValue,
    uiCount         => uiHeaderCount,
    bInvalidGrammar => bHeaderInvalid
    );
IF bHeaderInvalid THEN
    M_SetError(
        eNewError := E_HttpError.iInvalidHeader,
        sMessage  := 'request header grammar is invalid'
        );
    RETURN;
ELSIF NOT bHeaderFound THEN
    M_SetError(
        eNewError := E_HttpError.iMissingHost,
        sMessage  := 'request host is missing'
        );
    RETURN;
ELSIF uiHeaderCount > 1 THEN
    M_SetError(
        eNewError := E_HttpError.iDuplicateHost,
        sMessage  := 'request host is duplicated'
        );
    RETURN;
ELSE
    stRequest.sHost := sHeaderValue;
END_IF

bHeaderFound := F_HttpFindHeader(
    sHeaders        := sHeaderText,
    sHeaderName     := 'Content-Type',
    sValue          => sHeaderValue,
    uiCount         => uiHeaderCount,
    bInvalidGrammar => bHeaderInvalid

IF bHeaderFound THEN

END_IF

bHeaderFound := F_HttpFindHeader(
    sHeaders        := sHeaderText,
    sHeaderName     := 'Authorization',
    sValue          => sHeaderValue,
    uiCount         => uiHeaderCount,
    bInvalidGrammar => bHeaderInvalid
    );
IF bHeaderFound THEN
    stRequest.sAuthorization := sHeaderValue;
END_IF

stRequest.bTransferChunked := FALSE;
bHeaderFound := F_HttpFindHeader(
    sHeaders        := sHeaderText,
    sHeaderName     := 'Transfer-Encoding',
    sValue          => sHeaderValue,
    uiCount         => uiHeaderCount,
    bInvalidGrammar => bHeaderInvalid
    );
IF uiHeaderCount > 1 THEN
    M_SetError(
        eNewError := E_HttpError.iUnsupportedTransferEncoding,
        sMessage  := 'duplicate transfer-encoding'
        );
    RETURN;
END_IF
IF bHeaderFound THEN
    IF F_HttpAsciiEqualsIgnoreCase(
        sLeft  := sHeaderValue,
        sRight := 'chunked'

        stRequest.bTransferChunked := TRUE;
    ELSE
        M_SetError(
            eNewError := E_HttpError.iUnsupportedTransferEncoding,
            sMessage  := 'unsupported transfer-encoding'
            );
        RETURN;
    END_IF
END_IF

stRequest.bHasContentLength := FALSE;
bHeaderFound := F_HttpFindHeader(
    sHeaders        := sHeaderText,
    sHeaderName     := 'Content-Length',
    sValue          => sHeaderValue,
    uiCount         => uiHeaderCount,
    bInvalidGrammar => bHeaderInvalid
    );
IF uiHeaderCount > 1 THEN
    M_SetError(
        eNewError := E_HttpError.iInvalidContentLength,
        sMessage  := 'duplicate content-length'
        );
    RETURN;
END_IF
IF bHeaderFound THEN
    stRequest.bHasContentLength := TRUE;
    IF NOT F_HttpParseContentLength(
        sValue   := sHeaderValue,
        udiValue => udiContentLength
        ) THEN
        M_SetError(
            eNewError := E_HttpError.iInvalidContentLength,
            sMessage  := 'invalid content-length'
            );
        RETURN;
    END_IF
    stRequest.udiContentLength := udiContentLength;
    IF stRequest.udiContentLength > TO_UDINT(GVL_Http.cnMaxBodySize) THEN
        // 原因：声明长度已经超过固定 body 缓冲上限时，应立即报协议容量错误，不能等对端关闭后被 TCP 读错误覆盖。
        M_SetError(
            eNewError := E_HttpError.iBodyTooLarge,
            sMessage  := 'request content-length exceeds body limit'
            );
        RETURN;
    END_IF
END_IF

IF stRequest.bTransferChunked AND stRequest.bHasContentLength THEN
    M_SetError(
        eNewError := E_HttpError.iTransferEncodingContentLength,
        sMessage  := 'transfer-encoding and content-length coexist'
        );
    RETURN;
END_IF

stRequest.bConnectionClose := FALSE;
bHeaderFound := F_HttpFindHeader(
    sHeaders        := sHeaderText,
    sHeaderName     := 'Connection',
    sValue          => sHeaderValue,
    uiCount         => uiHeaderCount,
    bInvalidGrammar => bHeaderInvalid
    );
IF bHeaderFound THEN

        sLeft  := sHeaderValue,
        sRight := 'close'
        );
END_IF

iBodyStart := iHeaderEnd + 4;
iBodyLen := LEN(sMessage) - iBodyStart + 1;
IF iBodyLen < 0 THEN
    iBodyLen := 0;
END_IF
IF iBodyLen > GVL_Http.cnMaxBodySize THEN
    M_SetError(
        eNewError := E_HttpError.iBodyTooLarge,
        sMessage  := 'request body exceeds limit'
        );
    RETURN;
END_IF
IF iBodyLen > 0 THEN
    sBodyText := MID(sMessage, iBodyLen, iBodyStart);
ELSE
    sBodyText := '';
END_IF

IF stRequest.bTransferChunked THEN
    fbChunked.M_Decode(
        sChunked := sBodyText,
        sBody    => stRequest.sBody
        );
    IF fbChunked.bError THEN
        M_SetError(
            eNewError := fbChunked.eError,
            sMessage  := fbChunked.sDiagMsg
            );
        RETURN;
    END_IF

    stRequest.udiContentLength := udiBodyLength;
ELSIF stRequest.bHasContentLength THEN
    IF TO_UDINT(iBodyLen) < stRequest.udiContentLength THEN
        M_SetError(
            eNewError := E_HttpError.iNeedMoreData,
            sMessage  := 'request body is incomplete'

        RETURN;
    END_IF
    IF stRequest.udiContentLength > 0 THEN
        stRequest.sBody := LEFT(sBodyText, TO_INT(stRequest.udiContentLength));
    END_IF
    udiBodyLength := stRequest.udiContentLength;
ELSE
    stRequest.sBody := '';

END_IF

FUNCTION F_HttpAsciiEqualsIgnoreCase : BOOL

    sLeft  : STRING(255) := '';    // 诊断或协议文本字段。
    sRight : STRING(255) := '';    // 诊断或协议文本字段。
END_VAR

    uiLen   : UINT;               // 计数、长度或状态数值。
    byLeft  : BYTE;               // 计数、长度或状态数值。

    pRight  : POINTER TO BYTE;    // 当前扫描周期使用的字节指针。
END_VAR

// === IMPLEMENTATION ===
/// =======================================================================
/// 名称      : F_HttpAsciiEqualsIgnoreCase
/// 功能      : 按 HTTP Header 的 ASCII 语义进行大小写不敏感比较。
/// 库依赖    : 暂无
/// =======================================================================
/// 使用说明  : 1. 本函数只处理 ASCII 范围，避免引入 Unicode 或区域设置行为。
///           : 2. 长度不同直接返回 FALSE，适合 Header 名和固定 token 比较。
/// =======================================================================
// 工程说明：本段集中处理状态、边界或诊断，避免跨周期残留。
// 边界说明：执行前后保持输出和错误码可被在线诊断追踪。
F_HttpAsciiEqualsIgnoreCase := FALSE;

IF LEN(sLeft) <> LEN(sRight) THEN
    RETURN;
END_IF

uiLen := TO_UINT(LEN(sLeft));
IF uiLen = 0 THEN
    F_HttpAsciiEqualsIgnoreCase := TRUE;
    RETURN;
END_IF

pLeft := ADR(sLeft);
pRight := ADR(sRight);
FOR uiIndex := 0 TO uiLen - 1 DO
    byLeft := pLeft[uiIndex];
    byRight := pRight[uiIndex];
    IF (byLeft >= 16#61) AND (byLeft <= 16#7A) THEN
        byLeft := UINT_TO_BYTE(BYTE_TO_UINT(byLeft) - 32);
    END_IF
    IF (byRight >= 16#61) AND (byRight <= 16#7A) THEN
        byRight := UINT_TO_BYTE(BYTE_TO_UINT(byRight) - 32);
    END_IF
    IF byLeft <> byRight THEN
        RETURN;
    END_IF
END_FOR

/// 功能    : 严格解析 HTTP Content-Length。
/// 库依赖 : 暂无
{attribute 'hide_all_locals'}
FUNCTION F_HttpParseContentLength : BOOL
VAR_INPUT
    sValue : STRING(64) := '';    // 诊断或协议文本字段。
END_VAR
VAR_OUTPUT
    udiValue : UDINT;    // 计数、长度或状态数值。
END_VAR
VAR
    uiIndex : UINT;               // 计数、长度或状态数值。
    uiLen   : UINT;               // 计数、长度或状态数值。
    byChar  : BYTE;               // 计数、长度或状态数值。
    pText   : POINTER TO BYTE;    // 当前扫描周期使用的字节指针。
END_VAR

// === IMPLEMENTATION ===
/// =======================================================================
/// 名称      : F_HttpParseContentLength
/// 功能      : 只接受十进制非负整数并输出字节长度。
/// 库依赖    : 暂无
/// =======================================================================
/// 使用说明  : 1. 空值、非数字和值超过 UDINT 范围均返回 FALSE。
///           : 2. body 上限由 Parser 统一判断为 iBodyTooLarge，避免把合法长度字段误报为非法语法。
/// =======================================================================
// 工程说明：本段集中处理状态、边界或诊断，避免跨周期残留。
// 边界说明：执行前后保持输出和错误码可被在线诊断追踪。
F_HttpParseContentLength := FALSE;
udiValue := 0;
uiLen := TO_UINT(LEN(sValue));
IF uiLen = 0 THEN
    RETURN;
END_IF

pText := ADR(sValue);
FOR uiIndex := 0 TO uiLen - 1 DO
    byChar := pText[uiIndex];
    IF (byChar < 16#30) OR (byChar > 16#39) THEN
        RETURN;
    END_IF
    // 原因：Content-Length 字段本身只校验十进制语法和 UDINT 上限，超过库 body 上限由 Parser 分类诊断。
    IF (udiValue > 429496729) OR ((udiValue = 429496729) AND (BYTE_TO_UDINT(byChar - 16#30) > 5)) THEN
        RETURN;
    END_IF
    udiValue := (udiValue * 10) + BYTE_TO_UDINT(byChar - 16#30);
END_FOR

F_HttpParseContentLength := TRUE;
